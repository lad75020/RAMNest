cmake_minimum_required(VERSION 3.20)
set(CMAKE_OSX_DEPLOYMENT_TARGET "27.0" CACHE STRING "Minimum macOS" FORCE)
project(RAMNestNativeTmpfs LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(ramnest-helper NativeTmpfsHelper.cpp)
target_compile_options(ramnest-helper PRIVATE -Wall -Wextra -Werror)
# Only macOS system libraries. No FUSE SDK, MemoryFS, compression or rpath.
enable_testing()
add_executable(ramnest-native-protocol-tests NativeTmpfsProtocolTests.cpp)
target_compile_options(ramnest-native-protocol-tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME native-tmpfs-bounded-protocol COMMAND ramnest-native-protocol-tests)
